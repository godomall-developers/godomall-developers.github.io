+     *
+     * 한 컬럼에 고도몰 자체 상태와 우체국 배송상태가 함께 실리므로 둘을 차례로 판정함

+     * @return string|null 병기 대상이 아니면 null

+    public function getProgressLabel(?string $returnInvoiceStatus): ?string
+    {

+            return null;
+        }

+        if ($returnInvoiceStatus === EpostReturnInvoiceStatus::FAILED->value) {
+            return __('수거신청 실패');
+        }

+        if ($returnInvoiceStatus === EpostReturnInvoiceStatus::REQUESTED->value) {
+            return __('수거 접수');
+        }

+            EpostDeliveryStatus::COLLECT        => __('수거 완료'),
+            EpostDeliveryStatus::NOT_COLLECT    => __('미수거'),
+            EpostDeliveryStatus::TAKE_OVER,
+            EpostDeliveryStatus::SEND,
+            EpostDeliveryStatus::RECEIVE,
+            EpostDeliveryStatus::DELIVERY_READY => __('반송중'),
+            EpostDeliveryStatus::NOT_DELIVERY   => __('미배달'),
+            default                             => null,
+        };

+     * @param string|null $returnInvoiceNo     반품 수거 송장번호
+     * @param string|null $returnInvoiceStatus 반품 수거 상태

+    public function isTraceVisible(?string $returnInvoiceNo, ?string $returnInvoiceStatus): bool

+     * @param string|null $returnInvoiceStatus 반품 수거 상태

+     * @return array [orderGoodsNo => 서버 조립 URL] — 노출 대상 아닌 행은 뺌
+     */

+    {
+        $traceUrlMap = [];
+        $templatesByCarrier = [];
+
+        foreach ($rows as $row) {
+            if (!$this->isTraceVisible($row['returnInvoiceNo'], $row['returnInvoiceStatus']) || empty($row['returnCarrierNo'])) {
+                continue;
+            }
+
+            $carrierSno = (int) $row['returnCarrierNo'];
+            if (!array_key_exists($carrierSno, $templatesByCarrier)) {
+                $templatesByCarrier[$carrierSno] = $this->delivery->getDeliveryTraceUrlTemplate($carrierSno);
+            }
+
+            $template = $templatesByCarrier[$carrierSno];
+            $traceUrl = str_replace('__INVOICENO__', str_replace('-', '', $row['returnInvoiceNo']), (string) $template['traceUrl']);
+
+            if (!filter_var($traceUrl, FILTER_VALIDATE_URL)) {
+                $this->logger->warning('반송추적 URL 조립 실패 — 택배사 traceUrl 미설정', [
+                    'orderGoodsNo' => $row['orderGoodsNo'],
+                    'returnCarrierNo' => $carrierSno,
+                ]);
+                continue;
+            }
+
+            $traceUrlMap[$row['orderGoodsNo']] = $traceUrl;
+        }
+
+        return $traceUrlMap;
+    }
+
+    /**
+     * 상품주문번호별 병기 라벨·반송추적 URL — 목록이 조인하지 않는 화면용 보충 조회 1회
+     *
+     * @param array $orderNos       주문번호 목록
+     * @param array $orderGoodsSnos 상품주문번호 목록
+     * @return array [orderGoodsNo => ['progressLabel' => 라벨, 'traceUrl' => URL]] — 대상 아닌 값은 null
+     */
+    public function getDisplayMap(array $orderNos, array $orderGoodsSnos): array
+    {
+        if (empty($orderNos) || empty($orderGoodsSnos)) {
+            return [];
+        }
+
+        $rows = $this->returnInvoiceRepository->findReturnInvoiceStatusRows($orderNos, $orderGoodsSnos);
+        $traceUrlMap = $this->getTraceUrlMap($rows);
+
+        $displayMap = [];
+        foreach ($rows as $row) {
+            $displayMap[$row['orderGoodsNo']] = [
+                'progressLabel' => $this->getProgressLabel($row['returnInvoiceStatus']),
+                'traceUrl'      => $traceUrlMap[$row['orderGoodsNo']] ?? null,
+            ];
+        }
+
+        return $displayMap;
+    }
+
+    /**
+     * 마이페이지 주문 목록 행에 반송추적 URL 주입 — 보충 조회 1회, 노출 대상 아니면 null (템플릿은 null 여부로만 분기)
+     *
+     * @param array $orderData 주문 목록 (각 행에 orderNo·goods)
+     * @return array returnTraceUrl 이 주입된 주문 목록
+     */
+    public function assignTraceUrls(array $orderData): array
+    {
+        $orderGoodsSnos = array_column(array_merge(...array_column($orderData, 'goods')), 'sno');
+        if (empty($orderGoodsSnos)) {
+            return $orderData;
+        }
+
+        $traceUrlMap = $this->getTraceUrlMap($this->returnInvoiceRepository->findReturnInvoiceStatusRows(
+            array_values(array_unique(array_column($orderData, 'orderNo'))),
+            $orderGoodsSnos
+        ));
+
+        foreach ($orderData as &$order) {
+            foreach ($order['goods'] ?? [] as $goodsKey => $orderGoods) {
+                $order['goods'][$goodsKey]['returnTraceUrl'] = $traceUrlMap[$orderGoods['sno']] ?? null;
+            }
+        }
+
+        return $orderData;
+    }

+     * @param array $orderGoodsSnos 상품주문번호 목록
+     * @return array 통과한 상품주문번호 목록
+     */
+    public function findTargetSnos(array $orderGoodsSnos): array
+    {
+        return $this->buildTargetQuery(null)
+            ->whereIn('og.sno', array_map('intval', $orderGoodsSnos))
+            ->distinct()
+            ->pluck('og.sno')
+            ->all();
+    }
+
+    /**
+     * 신청 sno(og.userHandleSno) 기준으로 대상 조건을 통과하는 상품주문번호를 돌려줌 — 목록과 같은 대상 조건을 지남
+     *
+     * @param array $userHandleSnos 고객 신청 sno 목록
+     * @return array 통과한 상품주문번호 목록
+     */
+    public function findTargetSnosByUserHandleSnos(array $userHandleSnos): array
+    {
+        return $this->buildTargetQuery(null)
+            ->whereIn('og.userHandleSno', $userHandleSnos)
+            ->distinct()
+            ->pluck('og.sno')
+            ->all();
+    }
+
+    public function countTargets(EpostReturnSearchDTO $search): int
+    {
+        // oi.orderInfoCd=1 행이 한 주문에 여럿일 수 있어 주문상품 기준으로 센다
+        return $this->buildTargetQuery($search)->distinct()->count('og.sno');
+    }
+
+    /**
+     * 수거신청 대상 술어 — 목록·건수·게이트 재검증이 전부 이 한 지점을 지남
+     *
+     * @param ?EpostReturnSearchDTO $search 화면 검색 조건 — null 이면 술어만 적용함
+     */
+    protected function buildTargetQuery(?EpostReturnSearchDTO $search): Builder
+    {
+        $query = OrderGoods::query()
+            ->from(DB_ORDER_GOODS . ' AS og')
+            ->join(DB_ORDER . ' AS o', 'o.orderNo', '=', 'og.orderNo')
+            ->join(DB_ORDER_INFO . ' AS oi', function ($join) {
+                $join->on('oi.orderNo', '=', 'og.orderNo')
+                    ->where('oi.orderInfoCd', 1);
+            })
+            ->join(DB_ORDER_HANDLE . ' AS oh', function ($join) {
+                $join->on('oh.sno', '=', 'og.handleSno')
+                    ->on('oh.orderNo', '=', 'og.orderNo');
+            })
+            ->leftJoin(DB_ORDER_RETURN_INVOICE . ' AS rt', 'rt.orderGoodsNo', '=', 'og.sno')
+            ->leftJoin(DB_ORDER_DELIVERY . ' AS od', 'od.sno', '=', 'og.orderDeliverySno')
+            // 배송번호 = 상품별 배송비 코드(배송비조건 sno) — 삭제된 코드는 공란 처리라 존재 확인 조인을 거침
+            ->leftJoin(DB_SCM_DELIVERY_BASIC . ' AS sdb', 'sdb.sno', '=', 'od.deliverySno')
+            ->leftJoin(DB_SCM_MANAGE . ' AS sm', 'sm.scmNo', '=', 'og.scmNo')
+            ->leftJoin(DB_MEMBER . ' AS m', 'm.memNo', '=', 'o.memNo')
+            ->leftJoin(DB_GOODS . ' AS g', 'g.goodsNo', '=', 'og.goodsNo')
+            ->leftJoin(DB_GOODS_IMAGE . ' AS gi', function ($join) {
+                $join->on('gi.goodsNo', '=', 'og.goodsNo')
+                    ->where('gi.imageKind', 'list');
+            })
+            ->where('og.orderStatus', OrderStatusCode::BACK_REQUEST->value)
+            ->where('og.invoiceCompanySno', DEFAULT_CODE_GODOPOST)
+            ->whereRaw("TRIM(og.invoiceNo) != ''")
+            ->where('o.orderChannelFl', '!=', OrderChannel::NAVERPAY->value)
+            ->where(function ($where) {
+                $where->whereNull('rt.returnInvoiceStatus')
+                    ->orWhere('rt.returnInvoiceStatus', EpostReturnInvoiceStatus::FAILED->value);
+            });
+
+        if ($search !== null) {
+            $this->applyStatusFilter($query, $search);
+            $this->applyScmFilter($query, $search);
+            $this->applyDateFilter($query, $search);
+            $this->applyKeywordFilter($query, $search);
+        }
+
+        return $query;
+    }
+
+    protected function applyStatusFilter(Builder $query, EpostReturnSearchDTO $search): void
+    {
+        if ($search->isPendingOnly()) {
+            $query->whereNull('rt.returnInvoiceStatus');
+        } elseif ($search->isFailedOnly()) {
+            $query->where('rt.returnInvoiceStatus', EpostReturnInvoiceStatus::FAILED->value);
+        }
+    }
+
+    protected function applyScmFilter(Builder $query, EpostReturnSearchDTO $search): void
+    {
+        $headquartersOnly = $search->getHeadquartersOnly();
+
+        if ($headquartersOnly === true) {
+            $query->where('og.scmNo', DEFAULT_CODE_SCMNO);
+        } elseif ($headquartersOnly === false) {
+            $query->where('og.scmNo', '!=', DEFAULT_CODE_SCMNO);
+        } elseif (!empty($search->getScmNos())) {
+            $query->whereIn('og.scmNo', $search->getScmNos());
+        }
+    }
+
+    protected function applyDateFilter(Builder $query, EpostReturnSearchDTO $search): void
+    {
+        $column = $search->getDateColumn();
+        if ($column === null) {
+            return;
+        }
+
+        if ($search->getHandleMode() !== null) {
+            $query->where('oh.handleMode', $search->getHandleMode());
+        }
+
+        // 종료일이 YYYY-MM-DD만 오면 00:00:00으로 해석되어 종료일 당일 데이터가 전부 빠지므로 하루 끝까지 포함시킴
+        $query->whereBetween($column, [$search->getDateStart(), $search->getDateEnd() . ' 23:59:59']);
+    }
+
+    protected function applyKeywordFilter(Builder $query, EpostReturnSearchDTO $search): void
+    {
+        $column = $search->getSearchColumn();
+        if ($column === null) {
+            return;
+        }
+
+        if ($search->isLikeMatch()) {
+            $query->where($column, 'LIKE', '%' . addcslashes($search->getKeyword(), '%_\\') . '%');
+        } else {
+            $query->where($column, $search->getKeyword());
+        }
+    }
+}
