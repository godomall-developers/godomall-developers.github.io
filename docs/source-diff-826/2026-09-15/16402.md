+<?php

+namespace Bundle\Util\Order;
+
+/**
+ * [데드락 회피] 주문상품 잠금 순서 통일용 정렬 유틸
+ *
+ * es_memberOrderGoodsCount 는 (memNo, goodsNo) 기준으로 잠기므로, 여러 상품을 갱신할 때
+ * goodsNo 오름차순으로 처리해 트랜잭션 간 잠금 순서를 통일한다(데드락 패턴 A 예방)
+ * Order / OrderNew 양쪽에서 동일 로직을 공용으로 사용하기 위해 분리
+ */
+class OrderGoodsLockSortUtil
+{
+    /**
+     * 주문상품 sno 목록을 실제 주문상품 정보로 해석한 뒤 goodsNo 오름차순으로 정렬해 반환
+     *
+     * sno 입력 순서에 의존하지 않으며, 같은 goodsNo 의 상대 순서는 원본 순서를 보존
+     * (정렬 안정성 → 중복 goodsNo 누적 처리 동작 불변). goodsNo 를 해석하지 못한 항목은 0 으로 앞에 정렬
+     *
+     * @param object $order             getOnlyOrderGoodsData(orderNo, sno) 를 제공하는 주문 객체(Order|OrderNew)
+     * @param string $orderNo           주문번호
+     * @param array  $orderGoodsSnoList 주문상품 sno 목록
+     *
+     * @return array 각 항목이 array('seq'=>입력순번, 'goodsNo'=>정렬키, 'data'=>getOnlyOrderGoodsData 결과)
+     *               이며 goodsNo 오름차순으로 정렬된 목록
+     */
+    public static function resolveOrderGoodsSortedByGoodsNo(object $order, string $orderNo, array $orderGoodsSnoList): array
+    {
+        // 재사용 대비 방어: getOnlyOrderGoodsData 를 제공하지 않는 객체면 빈 목록 반환
+        if (!is_object($order) || !method_exists($order, 'getOnlyOrderGoodsData')) {
+            return [];
+        }
+
+        $resolvedList = [];
+        foreach ($orderGoodsSnoList as $seq => $orderGoodsSno) {
+            $orderGoodsData = $order->getOnlyOrderGoodsData($orderNo, $orderGoodsSno);
+            $resolvedList[] = [
+                'seq'     => $seq,
+                'goodsNo' => (is_array($orderGoodsData) && isset($orderGoodsData['goodsNo'])) ? (int) $orderGoodsData['goodsNo'] : 0,
+                'data'    => $orderGoodsData,
+            ];
+        }
+        usort($resolvedList, function ($leftGoods, $rightGoods) {
+            if ($leftGoods['goodsNo'] === $rightGoods['goodsNo']) {
+                return $leftGoods['seq'] - $rightGoods['seq']; // 동률은 원본 순서 유지(정렬 안정성 보장)
+            }
+
+            return $leftGoods['goodsNo'] - $rightGoods['goodsNo'];
+        });
+
+        return $resolvedList;
+    }
+}
